```rust
pub fn truncate_normalize(v: &[f32], dim: usize) -> Option<Vec<f32>> {
    if dim == 0 || dim > v.len() { return None; }
    let t = &v[..dim];
    if t.iter().any(|x| !x.is_finite()) { return None; }
    let norm = t.iter().map(|x| x * x).sum::<f32>().sqrt();
    if norm <= 1e-12 { return None; }
    Some(t.iter().map(|x| x / norm).collect())
}

pub fn quantize_i8(v: &[f32]) -> (Vec<i8>, f32) {
    let m = v.iter().fold(0f32, |a, x| a.max(x.abs()));
    if m == 0.0 { return (vec![0; v.len()], 0.0); }
    let s = 127.0 / m;
    (v.iter().map(|x| (x * s).round().clamp(-127.0, 127.0) as i8).collect(), m / 127.0)
}

pub fn dot_i8(a: &[i8], b: &[i8]) -> i32 {
    a.iter().zip(b).map(|(x, y)| *x as i32 * *y as i32).sum()
}

pub fn sign_bits(v: &[f32]) -> Vec<u64> {
    let mut out = vec![0u64; v.len().div_ceil(64)];
    for (i, x) in v.iter().enumerate() { if *x > 0.0 { out[i / 64] |= 1u64 << (i % 64); } }
    out
}

pub fn hamming(a: &[u64], b: &[u64]) -> u32 {
    a.iter().zip(b).map(|(x, y)| (x ^ y).count_ones()).sum()
}

pub fn asym_score(q: &[f32], bits: &[u64]) -> f32 {
    q.iter().enumerate()
        .map(|(i, x)| if (bits[i / 64] >> (i % 64)) & 1 == 1 { *x } else { -*x })
        .sum()
}

pub fn cosine(a: &[f32], b: &[f32]) -> f32 {
    let d: f32 = a.iter().zip(b).map(|(x, y)| x * y).sum();
    let na = a.iter().map(|x| x * x).sum::<f32>().sqrt();
    let nb = b.iter().map(|x| x * x).sum::<f32>().sqrt();
    if na == 0.0 || nb == 0.0 { 0.0 } else { d / (na * nb) }
}
```

```rust
pub struct ModelProfile {
```

```rust
use embed::vector::{asym_score, dot_i8, hamming, quantize_i8, sign_bits, truncate_normalize};

pub struct FlatIndex {
    dim: usize,
    bit_dim: usize,
    words: usize,
    codes: Vec<i8>,
    inv_scales: Vec<f32>,
    bits: Vec<u64>,
}

impl FlatIndex {
    pub fn new(dim: usize, bit_dim: usize) -> Self {
        Self { dim, bit_dim, words: bit_dim.div_ceil(64), codes: vec![], inv_scales: vec![], bits: vec![] }
    }
    pub fn len(&self) -> usize { self.inv_scales.len() }
    pub fn is_empty(&self) -> bool { self.len() == 0 }

    pub fn push(&mut self, full: &[f32]) -> bool {
        if full.len() < self.dim.max(self.bit_dim) { return false; }
        let (Some(v), Some(b)) = (truncate_normalize(full, self.dim), truncate_normalize(full, self.bit_dim)) else { return false };
        let (c, s) = quantize_i8(&v);
        self.codes.extend_from_slice(&c);
        self.inv_scales.push(s);
        self.bits.extend_from_slice(&sign_bits(&b));
        true
    }

    fn code(&self, i: usize) -> &[i8] { &self.codes[i * self.dim..(i + 1) * self.dim] }
    fn bit(&self, i: usize) -> &[u64] { &self.bits[i * self.words..(i + 1) * self.words] }

    fn rescore(&self, q: &[f32], ids: impl Iterator<Item = usize>, k: usize) -> Vec<(u32, f32)> {
        let Some(qn) = truncate_normalize(q, self.dim) else { return vec![] };
        let (qc, qs) = quantize_i8(&qn);
        let mut scored: Vec<(u32, f32)> = ids
            .map(|i| (i as u32, dot_i8(&qc, self.code(i)) as f32 * qs * self.inv_scales[i]))
            .collect();
        top_k(&mut scored, k);
        scored
    }

    pub fn search_exact(&self, q: &[f32], k: usize) -> Vec<(u32, f32)> {
        self.rescore(q, 0..self.len(), k)
    }

    pub fn search_hamming(&self, q: &[f32], candidates: usize, k: usize) -> Vec<(u32, f32)> {
        let Some(qb) = truncate_normalize(q, self.bit_dim) else { return vec![] };
        let qbits = sign_bits(&qb);
        let mut cand: Vec<(u32, f32)> = (0..self.len()).map(|i| (i as u32, -(hamming(&qbits, self.bit(i)) as f32))).collect();
        top_k(&mut cand, candidates);
        self.rescore(q, cand.into_iter().map(|c| c.0 as usize), k)
    }

    pub fn search_asym(&self, q: &[f32], candidates: usize, k: usize) -> Vec<(u32, f32)> {
        let Some(qb) = truncate_normalize(q, self.bit_dim) else { return vec![] };
        let mut cand: Vec<(u32, f32)> = (0..self.len()).map(|i| (i as u32, asym_score(&qb, self.bit(i)))).collect();
        top_k(&mut cand, candidates);
        self.rescore(q, cand.into_iter().map(|c| c.0 as usize), k)
    }
}

/// 점수 내림차순 상위 k개만 남긴다(동점은 id 오름차순).
fn top_k(v: &mut Vec<(u32, f32)>, k: usize) {
    let cmp = |a: &(u32, f32), b: &(u32, f32)| b.1.total_cmp(&a.1).then(a.0.cmp(&b.0));
    if v.len() > k && k > 0 { v.select_nth_unstable_by(k - 1, cmp); v.truncate(k); }
    if k == 0 { v.clear(); }
    v.sort_by(cmp);
}
```

```rust
#[derive(Debug, thiserror::Error)]
```

```rust
    #[error("모델 파일이 없습니다: {0}")] ModelNotFound(String),
    #[error("모델을 불러오지 못했습니다: {0}")] ModelLoad(String),
    #[error("추론 컨텍스트를 만들지 못했습니다")] Context,

    #[error("추론 실패(code {0})")] Decode(i32),

    BACKEND.get_or_init(|| unsafe {
        if let Some(d) = dir {
            let c = CString::new(d.to_string_lossy().as_bytes()).map_err(|e| e.to_string())?;

        }
        llama_backend_init();
        Ok(())
    }).clone().map_err(EmbedError::Backend)
}

pub fn system_info() -> String {
    unsafe { CStr::from_ptr(llama_print_system_info()).to_string_lossy().into_owned() }

pub struct Embedder {
    model: *mut llama_model,
    ctx: *mut llama_context,
    vocab: *const llama_vocab,
    n_embd: usize,
    max_tokens: usize,
    encoder_only: bool,
}
unsafe impl Send for Embedder {}

impl Embedder {
    pub fn load(model_path: &Path, opts: &EngineOptions) -> Result<Self, EmbedError> {
        if !model_path.is_file() { return Err(EmbedError::ModelNotFound(model_path.display().to_string())); }
        init_backend(opts.backend_dir.as_deref())?;
        let cpath = CString::new(model_path.to_string_lossy().as_bytes()).map_err(|_| EmbedError::ModelNotFound(model_path.display().to_string()))?;
        unsafe {
            let mut mp = llama_model_default_params();
            mp.use_mmap = true;
            let model = llama_model_load_from_file(cpath.as_ptr(), mp);
            if model.is_null() { return Err(EmbedError::ModelLoad(model_path.display().to_string())); }
            let mut cp = llama_context_default_params();
            cp.embeddings = true;
            cp.pooling_type = llama_pooling_type_LLAMA_POOLING_TYPE_MEAN;
            cp.n_ctx = opts.max_tokens as u32;
            cp.n_batch = opts.max_tokens as u32;
            cp.n_ubatch = opts.max_tokens as u32;
            cp.n_seq_max = 8;
            cp.n_threads = opts.threads as i32;
            cp.n_threads_batch = opts.threads as i32;
            let ctx = llama_init_from_model(model, cp);
            if ctx.is_null() { llama_model_free(model); return Err(EmbedError::Context); }
            let encoder_only = llama_model_has_encoder(model) && !llama_model_has_decoder(model);
            Ok(Self {
                model, ctx, vocab: llama_model_get_vocab(model),
                n_embd: llama_model_n_embd(model) as usize,
                max_tokens: opts.max_tokens, encoder_only,
            })
        }
    }

    pub fn n_embd(&self) -> usize { self.n_embd }

    fn tokenize(&self, text: &str) -> Result<Vec<llama_token>, EmbedError> {
        let mut buf = vec![0 as llama_token; text.len() + 8];
        let mut n = unsafe { llama_tokenize(self.vocab, text.as_ptr() as *const c_char, text.len() as i32, buf.as_mut_ptr(), buf.len() as i32, true, false) };
        if n < 0 {
            buf.resize((-n) as usize, 0);
            n = unsafe { llama_tokenize(self.vocab, text.as_ptr() as *const c_char, text.len() as i32, buf.as_mut_ptr(), buf.len() as i32, true, false) };
            if n < 0 { return Err(EmbedError::Tokenize); }
        }
        buf.truncate(n as usize);
        Ok(buf)
    }

    pub fn token_count(&self, text: &str) -> Result<usize, EmbedError> { Ok(self.tokenize(text)?.len()) }

    pub fn embed(&mut self, inputs: &[String]) -> Result<Vec<Embedding>, EmbedError> {
        let mut toks = Vec::with_capacity(inputs.len());
        for s in inputs {
            if s.trim().is_empty() { return Err(EmbedError::EmptyInput); }
            let mut t = self.tokenize(s)?;
            let truncated = t.len() > self.max_tokens;
            t.truncate(self.max_tokens);
            toks.push((t, truncated));
        }
        // 토큰 합이 max_tokens 이하가 되도록 묶어서 한 번에 처리(양방향 어텐션: 한 ubatch에 들어가야 함)
        let mut out = Vec::with_capacity(inputs.len());
        let mut i = 0;
        while i < toks.len() {
            let mut j = i; let mut total = 0;
            while j < toks.len() && j - i < 8 && total + toks[j].0.len() <= self.max_tokens { total += toks[j].0.len(); j += 1; }
            if j == i { j = i + 1; total = toks[i].0.len(); }
            out.extend(self.run_group(&toks[i..j], total)?);
            i = j;
        }
        Ok(out)
    }

    fn run_group(&mut self, group: &[(Vec<llama_token>, bool)], total: usize) -> Result<Vec<Embedding>, EmbedError> {
        unsafe {
            llama_memory_clear(llama_get_memory(self.ctx), true);
            let mut batch = llama_batch_init(total as i32, 0, 1);
            let mut k = 0usize;
            for (seq, (t, _)) in group.iter().enumerate() {
                for (pos, tok) in t.iter().enumerate() {
                    *batch.token.add(k) = *tok;
                    *batch.pos.add(k) = pos as i32;
                    *batch.n_seq_id.add(k) = 1;
                    *(*batch.seq_id.add(k)) = seq as i32;
                    *batch.logits.add(k) = 1;
                    k += 1;
                }
            }
            batch.n_tokens = k as i32;
            let rc = if self.encoder_only { llama_encode(self.ctx, batch) } else { llama_decode(self.ctx, batch) };
            if rc != 0 { llama_batch_free(batch); return Err(EmbedError::Decode(rc)); }
            let mut res = Vec::with_capacity(group.len());
            for (seq, (t, tr)) in group.iter().enumerate() {
                let p = llama_get_embeddings_seq(self.ctx, seq as i32);
                if p.is_null() { llama_batch_free(batch); return Err(EmbedError::Decode(-1)); }
                res.push(Embedding { vector: std::slice::from_raw_parts(p, self.n_embd).to_vec(), n_tokens: t.len(), truncated: *tr });
            }
            llama_batch_free(batch);
            Ok(res)
        }
    }
}

impl Drop for Embedder {
    fn drop(&mut self) { unsafe { llama_free(self.ctx); llama_model_free(self.model); } }
}
```

```rust
pub enum Container { Ole, Zip, Other([u8; 8]) }

pub fn sniff(b: &[u8]) -> Container {
    if b.starts_with(&[0xD0, 0xCF, 0x11, 0xE0, 0xA1, 0xB1, 0x1A, 0xE1]) { return Container::Ole; }
    if b.starts_with(b"PK\x03\x04") { return Container::Zip; }
    let mut m = [0u8; 8];
    for (i, x) in b.iter().take(8).enumerate() { m[i] = *x; }
    Container::Other(m)
}

pub fn magic_hex(b: &[u8]) -> String { b.iter().take(8).map(|x| format!("{x:02x}")).collect() }
```

```rust
use crate::magic::{sniff, magic_hex, Container};
use doc::{Block, BlockKind, Document};

#[derive(Debug, Clone, serde::Serialize, serde::Deserialize, PartialEq)]
pub enum FailKind { Encrypted, Corrupt, NotContainer, Unsupported }

pub struct Extracted { pub doc: Document, pub distribution: bool }

pub fn extract_hwp(bytes: &[u8]) -> Result<Extracted, (FailKind, String)> {
    if let Container::Other(_) = sniff(bytes) { return Err((FailKind::NotContainer, magic_hex(bytes))); }
    let parsed = rhwp::parse_document(bytes).map_err(|e| (classify(&e.to_string()), magic_hex(bytes)))?;
    let mut blocks = Vec::new();
    // 변환 규칙 (rhwp 실제 타입에 맞춰 구현):
    //  - 문단 → BlockKind::Paragraph (빈 문단 제외, 앞뒤 공백 trim)
    //  - 개요/제목 스타일 문단(스타일 이름에 "개요" 또는 "제목", 또는 rag가 heading으로 표시) → BlockKind::Heading
    //  - 표: 첫 행 → BlockKind::TableHeader("열1 | 열2 | ..."), 나머지 행 → BlockKind::TableRow("열1: 값1 | 열2: 값2")
    //  - 머리말·꼬리말·각주는 제외
    //  rhwp의 rag::build_chunks가 위 정보를 이미 제공하면 그 출력에서 블록을 만들어도 된다.
    walk(&parsed, &mut blocks);
    let title = blocks.iter().find(|b| b.kind == BlockKind::Heading).map(|b| b.text.clone());
    Ok(Extracted { doc: Document { title, blocks }, distribution: is_distribution(&parsed) })
}

fn classify(msg: &str) -> FailKind {
    let m = msg.to_lowercase();
    if m.contains("password") || m.contains("encrypt") || m.contains("암호") { FailKind::Encrypted }
    else if m.contains("unsupported") || m.contains("version") { FailKind::Unsupported }
    else { FailKind::Corrupt }
}
```

```rust
#[derive(Debug, Clone, serde::Serialize, serde::Deserialize)]
pub enum Outcome {
    Ok { doc: Document, distribution: bool },
    Failed { kind: FailKind, magic: String },
    Timeout,
    Crashed { code: Option<i32> },
}

pub fn run_isolated(exe: &Path, file: &Path, timeout: Duration) -> Outcome {
    let mut cmd = Command::new(exe);
    cmd.arg(file);
    run_command(cmd, timeout)
}

pub fn run_command(mut cmd: Command, timeout: Duration) -> Outcome {
    let mut child = match cmd.stdout(Stdio::piped()).stderr(Stdio::null()).spawn() {
        Ok(c) => c, Err(_) => return Outcome::Crashed { code: None },
    };
    let mut stdout = child.stdout.take().unwrap();
    let reader = std::thread::spawn(move || { let mut s = String::new(); let _ = stdout.read_to_string(&mut s); s });
    match child.wait_timeout(timeout) {
        Ok(Some(status)) => {
            let out = reader.join().unwrap_or_default();
            serde_json::from_str(out.trim()).unwrap_or(Outcome::Crashed { code: status.code() })
        }
        _ => { let _ = child.kill(); let _ = child.wait(); Outcome::Timeout }
    }
}
```

```rust
use doc::{BlockKind, Document};

pub struct ChunkParams { pub target_tokens: usize, pub max_tokens: usize, pub max_chunks: usize }

/// `text`를 토큰 한도 이하가 되도록 글자 경계에서 자른다(이진 탐색).
fn cut_to(text: &str, max_tokens: usize, count: &dyn Fn(&str) -> usize) -> String {
    if count(text) <= max_tokens { return text.to_string(); }
    let idx: Vec<usize> = text.char_indices().map(|(i, _)| i).chain([text.len()]).collect();
    let (mut lo, mut hi) = (0usize, idx.len() - 1);
    while lo < hi {
        let mid = (lo + hi + 1) / 2;
        if count(&text[..idx[mid]]) <= max_tokens { lo = mid } else { hi = mid - 1 }
    }
    text[..idx[lo]].to_string()
}

pub fn build_card(doc: &Document, max_tokens: usize, count: &dyn Fn(&str) -> usize) -> Option<String> {
    let mut parts: Vec<String> = Vec::new();
    if let Some(t) = doc.title.as_deref().map(str::trim).filter(|t| !t.is_empty()) { parts.push(t.to_string()); }
    for b in &doc.blocks {
        if b.kind == BlockKind::Heading {
            let t = b.text.trim();
            if !t.is_empty() && parts.first().map(String::as_str) != Some(t) { parts.push(t.to_string()); }
        }
    }
    if let Some(p) = doc.blocks.iter().find(|b| b.kind == BlockKind::Paragraph && !b.text.trim().is_empty()) {
        parts.push(p.text.trim().to_string());
    }
    if parts.is_empty() { return None; }
    let card = cut_to(&parts.join("\n"), max_tokens, count);
    (!card.trim().is_empty()).then_some(card)
}

/// 문장 경계(다. / . / 줄바꿈) 우선으로 한도 이하 조각으로 나눈다.
fn split_long(text: &str, max: usize, count: &dyn Fn(&str) -> usize) -> Vec<String> {
    let mut out = Vec::new();
    let mut rest = text.trim();
    while !rest.is_empty() {
        if count(rest) <= max { out.push(rest.to_string()); break; }
        let head = cut_to(rest, max, count);
        let cut = head.rfind(['.', '\n']).map(|i| i + 1).filter(|i| *i > head.len() / 2).unwrap_or(head.len());
        out.push(rest[..cut].trim().to_string());
        rest = rest[cut..].trim_start();
    }
    out
}

pub fn build_chunks(doc: &Document, p: &ChunkParams, count: &dyn Fn(&str) -> usize) -> Vec<String> {
    let mut chunks: Vec<String> = Vec::new();
    let mut cur: Vec<String> = Vec::new();
    let mut table_header: Option<String> = None;
    let flush = |cur: &mut Vec<String>, chunks: &mut Vec<String>| {
        if cur.iter().any(|s| !s.trim().is_empty()) { chunks.push(cur.join("\n")); }
        cur.clear();
    };
    for b in &doc.blocks {
        let text = b.text.trim();
        if text.is_empty() { continue; }
        match b.kind {
            BlockKind::Heading => { flush(&mut cur, &mut chunks); table_header = None; cur.push(text.to_string()); }
            BlockKind::TableHeader => { flush(&mut cur, &mut chunks); table_header = Some(text.to_string()); cur.push(text.to_string()); }
            BlockKind::Paragraph | BlockKind::TableRow => {
                if b.kind == BlockKind::Paragraph { table_header = None; }
                for piece in split_long(text, p.max_tokens.saturating_sub(table_header.as_deref().map(count).unwrap_or(0) + 1).max(1), count) {
                    let candidate = if cur.is_empty() { piece.clone() } else { format!("{}\n{}", cur.join("\n"), piece) };
                    if !cur.is_empty() && count(&candidate) > p.target_tokens {
                        flush(&mut cur, &mut chunks);
                        if let (BlockKind::TableRow, Some(h)) = (&b.kind, &table_header) { cur.push(h.clone()); }
                    }
                    cur.push(piece);
                }
            }
        }
        if chunks.len() >= p.max_chunks { break; }
    }
    flush(&mut cur, &mut chunks);
    chunks.truncate(p.max_chunks);
    chunks
}
```

```rust
use std::collections::HashSet;

pub fn ndcg_at_k(ranked: &[String], relevant: &HashSet<String>, k: usize) -> f64 {
    let dcg: f64 = ranked.iter().take(k).enumerate()
        .filter(|(_, d)| relevant.contains(*d)).map(|(i, _)| 1.0 / ((i + 2) as f64).log2()).sum();
    let idcg: f64 = (0..relevant.len().min(k)).map(|i| 1.0 / ((i + 2) as f64).log2()).sum();
    if idcg == 0.0 { 0.0 } else { dcg / idcg }
}

pub fn recall_at_k(ranked: &[String], relevant: &HashSet<String>, k: usize) -> f64 {
    if relevant.is_empty() { return 0.0; }
    ranked.iter().take(k).filter(|d| relevant.contains(*d)).count() as f64 / relevant.len() as f64
}
```

```rust
pub fn anon(p: &Path) -> String {
    let h = Sha256::digest(p.to_string_lossy().as_bytes());
    let hex: String = h.iter().take(6).map(|b| format!("{b:02x}")).collect();
    let ext = p.extension().map(|e| e.to_string_lossy().to_lowercase()).unwrap_or_default();

}
```
